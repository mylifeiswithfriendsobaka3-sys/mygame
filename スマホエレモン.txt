        // Base64画像読込処理

            'BRR': { 'ATK': -1, 'MAG': 1,  'BRR': 0, 'CHG': 1 },
            'CHG': { 'ATK': -1, 'MAG': -1, 'BRR': -1, 'CHG': 0 }
        };

        const cmdNames = { 'ATK': '攻撃', 'MAG': '魔法', 'BRR': 'バリア', 'CHG': 'チャージ' };
        const allCmdKeys = ['ATK', 'MAG', 'BRR', 'CHG'];

        let party = [];
        let activeIndex = 0;
        let enemy = {};
        let isBattleEnd = false;
        let lastPlayerCmd = null;
        let enemyLastCmd = null;
        let isFirstTurn = true;
        let selectedSpellType = null;
        let selectedPartyNames = [];

        const els = {
            pName: document.getElementById('p-name'), pAttr: document.getElementById('p-attr'),
            pHpBar: document.getElementById('p-hp-bar'), pHpText: document.getElementById('p-hp-text'),
            pMpText: document.getElementById('p-mp-text'),
            pStatsDetail: document.getElementById('p-stats-detail'), pStatus: document.getElementById('p-status'),
            partyList: document.getElementById('party-list'),
            eName: document.getElementById('e-name'), eAttr: document.getElementById('e-attr'),
            eHpBar: document.getElementById('e-hp-bar'), eHpText: document.getElementById('e-hp-text'),
            eMpText: document.getElementById('e-mp-text'),
            eStatsDetail: document.getElementById('e-stats-detail'), eStatus: document.getElementById('e-status'),
            log: document.getElementById('log-area'),
            commandArea: document.getElementById('command-area'),
            magicSelectArea: document.getElementById('magic-select-area'),
            magicButtons: document.getElementById('magic-buttons'),
            startSelectArea: document.getElementById('start-select-area'),
            characterGrid: document.getElementById('character-grid'),
            btnStartBattle: document.getElementById('btn-start-battle'),
            selectCount: document.getElementById('select-count'),
            switchArea: document.getElementById('switch-area'),
            switchButtons: document.getElementById('switch-buttons'),
            resetBtn: document.getElementById('reset-btn'),
            btns: {
                'ATK': document.getElementById('btn-ATK'), 'MAG': document.getElementById('btn-MAG'),
                'BRR': document.getElementById('btn-BRR'), 'CHG': document.getElementById('btn-CHG')
            }
        };

        function showStartSelect() {
            playSE('click');
            els.commandArea.style.display = 'none';
            els.magicSelectArea.style.display = 'none';
            els.switchArea.style.display = 'none';
            els.resetBtn.style.display = 'none';
            els.startSelectArea.style.display = 'block';

            selectedPartyNames = [];
            renderCharacterGrid();
        }

        function renderCharacterGrid() {
            els.characterGrid.innerHTML = '';
            els.selectCount.innerText = selectedPartyNames.length;
            els.btnStartBattle.disabled = selectedPartyNames.length !== 3;

            for (let name in masterMons) {
                const m = masterMons[name];
                const isSelected = selectedPartyNames.includes(name);
                const card = document.createElement('div');
                card.className = `select-card ${isSelected ? 'selected' : ''}`;
                card.innerHTML = `<div>${m.name}</div><div style="font-size:8px;color:#aaa;">${m.attrName}</div>`;
                card.onclick = () => toggleSelectCharacter(name);
                els.characterGrid.appendChild(card);
            }
        }

        function toggleSelectCharacter(name) {
            playSE('click');
            const idx = selectedPartyNames.indexOf(name);
            if (idx >= 0) {
                selectedPartyNames.splice(idx, 1);
            } else {
                if (selectedPartyNames.length < 3) {
                    selectedPartyNames.push(name);
                }
            }
            renderCharacterGrid();
        }

        function confirmPartyAndStart() {
            if (selectedPartyNames.length !== 3) return;
            playSE('click');

            const spellLv = parseInt(document.getElementById('magic-lv-select').value);

            party = selectedPartyNames.map(name => {
                const m = masterMons[name];
                let spellLvs = {};
                m.spells.forEach(s => spellLvs[s] = spellLv);
                return { ...m, hp: m.maxHp, mp: m.maxMp, charged: false, debuff: null, spellLevels: spellLvs };
            });

            const allNames = Object.keys(masterMons);
            const enemyName = allNames[Math.floor(Math.random() * allNames.length)];
            const eMaster = masterMons[enemyName];

            let eSpellLvs = {};
            eMaster.spells.forEach(s => eSpellLvs[s] = 3);
            enemy = { ...eMaster, hp: eMaster.maxHp, mp: eMaster.maxMp, charged: false, debuff: null, spellLevels: eSpellLvs };

            activeIndex = 0;
            isBattleEnd = false;
            lastPlayerCmd = null;
            enemyLastCmd = null;
            isFirstTurn = true;

            els.startSelectArea.style.display = 'none';
            els.commandArea.style.display = 'block';
            els.log.innerHTML = '';

            updateUi();
            addLog(`<span class='log-system'>バトル開始！野生の ${enemy.name} が現れた！</span>`);
        }

        function updateUi() {
            const curP = party[activeIndex];

            els.pName.innerText = curP.name;
            els.pAttr.innerText = curP.attrName;
            els.pAttr.className = `attribute ${curP.attrClass}`;
            els.pHpBar.style.width = (curP.hp / curP.maxHp * 100) + '%';
            els.pHpText.innerText = `HP: ${curP.hp}/${curP.maxHp}`;
            els.pMpText.innerText = `MP: ${curP.mp}/${curP.maxMp}`;
            els.pStatsDetail.innerText = `ATK:${curP.atk} MATK:${curP.matk} SPD:${curP.spd} LUK:${curP.luk}`;

            let pStatusTxt = curP.charged ? '<span class="charged">[CHG]</span> ' : '';
            if (curP.debuff) pStatusTxt += `<span class="log-enemy">[${curP.debuff.name}]</span> `;
            els.pStatus.innerHTML = pStatusTxt;

            els.partyList.innerHTML = '';
            party.forEach((m, idx) => {
                let cls = 'party-icon';
                if (idx === activeIndex) cls += ' active';
                if (m.hp <= 0) cls += ' fainted';
                els.partyList.innerHTML += `<div class="${cls}">${m.name}</div>`;
            });

            els.eName.innerText = enemy.name;
            els.eAttr.innerText = enemy.attrName;
            els.eAttr.className = `attribute ${enemy.attrClass}`;
            els.eHpBar.style.width = (enemy.hp / enemy.maxHp * 100) + '%';
            els.eHpText.innerText = `HP: ${enemy.hp}/${enemy.maxHp}`;
            els.eMpText.innerText = `MP: ${enemy.mp}/${enemy.maxMp}`;
            els.eStatsDetail.innerText = `ATK:${enemy.atk} MATK:${enemy.matk} SPD:${enemy.spd} LUK:${enemy.luk}`;

            let eStatusTxt = enemy.charged ? '<span class="charged">[CHG]</span> ' : '';
            if (enemy.debuff) eStatusTxt += `<span class="log-enemy">[${enemy.debuff.name}]</span> `;
            els.eStatus.innerHTML = eStatusTxt;

            ['ATK', 'MAG', 'BRR', 'CHG'].forEach(cmd => {
                let disable = false;
                let titleText = "";
                if (cmd === lastPlayerCmd) { disable = true; titleText = "連続使用不可"; }
                if (cmd === 'CHG' && isFirstTurn) { disable = true; titleText = "初手使用不可"; }
                els.btns[cmd].disabled = disable;
                els.btns[cmd].title = titleText;
            });
        }

        function addLog(msg) {
            els.log.innerHTML += `<div>${msg}</div>`;
            els.log.scrollTop = els.log.scrollHeight;
        }

        function openMagicMenu() {
            playSE('click');
            const curP = party[activeIndex];
            els.magicButtons.innerHTML = '';

            curP.spells.forEach(spellKey => {
                const dDef = debuffDefs[spellKey];
                const btn = document.createElement('button');
                btn.className = 'command-btn btn-magic';
                btn.innerText = dDef.name;
                btn.onclick = () => selectMagicAndExecute(spellKey);
                els.magicButtons.appendChild(btn);
            });

            els.commandArea.style.display = 'none';
            els.magicSelectArea.style.display = 'block';
        }

        function closeMagicMenu() {
            playSE('click');
            els.magicSelectArea.style.display = 'none';
            els.commandArea.style.display = 'block';
        }

        function selectMagicAndExecute(spellKey) {
            selectedSpellType = spellKey;
            closeMagicMenu();
            playerAction('MAG');
        }

        function chooseEnemyAction() {
            let options = ['ATK', 'MAG', 'BRR', 'CHG'].filter(c => c !== enemyLastCmd);
            if (isFirstTurn) options = options.filter(c => c !== 'CHG');
            return options[Math.floor(Math.random() * options.length)];
        }

        function playerAction(pCmd) {
            if (isBattleEnd) return;

            let curP = party[activeIndex];
            let actualPCmd = pCmd;

            if (curP.debuff && curP.debuff.type === 'PARALYSIS' && Math.random() < 0.5) {
                addLog(`<span class='log-player'>${curP.name} は身体が痺れて動けない！</span>`);
                actualPCmd = null;
            } else if (curP.debuff && curP.debuff.type === 'CONFUSION' && Math.random() < 0.4) {
                actualPCmd = allCmdKeys[Math.floor(Math.random() * allCmdKeys.length)];
                addLog(`<span class='log-confused'>${curP.name} は混乱している！(コマンド判定: ${cmdNames[actualPCmd]})</span>`);
            }

            let eCmd = chooseEnemyAction();
            if (enemy.debuff && enemy.debuff.type === 'PARALYSIS' && Math.random() < 0.5) {
                addLog(`<span class='log-enemy'>${enemy.name} は身体が痺れて動けない！</span>`);
                eCmd = null;
            } else if (enemy.debuff && enemy.debuff.type === 'CONFUSION' && Math.random() < 0.4) {
                eCmd = allCmdKeys[Math.floor(Math.random() * allCmdKeys.length)];
                addLog(`<span class='log-confused'>${enemy.name} は混乱している！(敵コマンド: ${cmdNames[eCmd]})</span>`);
            }

            if (actualPCmd) playSE(actualPCmd.toLowerCase());

            executeTurn(actualPCmd, eCmd);

            lastPlayerCmd = actualPCmd;
            enemyLastCmd = eCmd;
            isFirstTurn = false;

            if (!isBattleEnd) updateUi();
        }

        function executeTurn(pCmd, eCmd) {
            const curP = party[activeIndex];

            if (!pCmd && !eCmd) {
                addLog("<span class='log-system'>両者ともに麻痺して動けなかった...</span>");
                return;
            }

            if (!pCmd) {
                processSingleAction(enemy, curP, eCmd, 'enemy');
                checkFaint();
                return;
            }

            if (!eCmd) {
                processSingleAction(curP, enemy, pCmd, 'player');
                checkFaint();
                return;
            }

            let result = rpsMap[pCmd][eCmd];

            if (result === 1) {
                addLog(`<span class='log-lucky'>【3すくみ勝利！】 ${curP.name}の[${cmdNames[pCmd]}] ＞ ${enemy.name}の[${cmdNames[eCmd]}]</span>`);
                processSingleAction(curP, enemy, pCmd, 'player', 1.5);
            } else if (result === -1) {
                addLog(`<span class='log-lose'>【3すくみ敗北...】 ${curP.name}の[${cmdNames[pCmd]}] ＜ ${enemy.name}の[${cmdNames[eCmd]}]</span>`);
                processSingleAction(enemy, curP, eCmd, 'enemy', 1.5);
            } else {
                addLog(`<span class='log-system'>【あいこ】 双方の[${cmdNames[pCmd]}]が激突！</span>`);
                if (curP.spd >= enemy.spd) {
                    processSingleAction(curP, enemy, pCmd, 'player');
                    if (enemy.hp > 0) processSingleAction(enemy, curP, eCmd, 'enemy');
                } else {
                    processSingleAction(enemy, curP, eCmd, 'enemy');
                    if (curP.hp > 0) processSingleAction(curP, enemy, pCmd, 'player');
                }
            }

            checkFaint();
        }

        function processSingleAction(attacker, defender, cmd, side, rpsBonus = 1.0) {
            const isPlayer = side === 'player';
            const cls = isPlayer ? 'log-player' : 'log-enemy';

            if (cmd === 'ATK') {
                let dmg = Math.max(1, Math.floor((attacker.atk * 1.5 - defender.matk * 0.2) * rpsBonus));
                if (attacker.charged) { dmg = Math.floor(dmg * 2.0); attacker.charged = false; }
                if (attacker.debuff && attacker.debuff.type === 'ATK_DOWN') dmg = Math.floor(dmg * 0.5);
                if (defender.debuff && defender.debuff.type === 'DEF_DOWN') dmg = Math.floor(dmg * 1.15);

                defender.hp = Math.max(0, defender.hp - dmg);
                triggerShake(isPlayer ? 'enemy' : 'player');
                addLog(`<span class='${cls}'>${attacker.name}の攻撃！ ${defender.name}に ${dmg} ダメージ！</span>`);
            } else if (cmd === 'MAG') {
                let spellKey = isPlayer ? selectedSpellType : attacker.spells[Math.floor(Math.random() * attacker.spells.length)];
                let spellLv = attacker.spellLevels[spellKey] || 3;
                let procRate = spellRates[spellLv];

                let dmg = Math.max(1, Math.floor((attacker.matk * 1.2) * rpsBonus));
                defender.hp = Math.max(0, defender.hp - dmg);
                triggerShake(isPlayer ? 'enemy' : 'player');

                let dDef = debuffDefs[spellKey];
                addLog(`<span class='${cls}'>${attacker.name}は魔法[${dDef.name}]を唱えた！ ${defender.name}に ${dmg} ダメージ！</span>`);

                if (Math.random() < procRate) {
                    defender.debuff = { type: spellKey, name: dDef.name };
                    addLog(`<span class='log-special'>追加効果発動！ ${defender.name} は [${dDef.name}] 状態になった！</span>`);
                }
            } else if (cmd === 'BRR') {
                let heal = Math.floor(attacker.maxHp * 0.15);
                attacker.hp = Math.min(attacker.maxHp, attacker.hp + heal);
                addLog(`<span class='${cls}'>${attacker.name}はバリアを展開！ HPが ${heal} 回復！</span>`);
            } else if (cmd === 'CHG') {
                attacker.charged = true;
                addLog(`<span class='${cls}'>${attacker.name}は力を溜めている！ 次の攻撃威力アップ！</span>`);
            }
        }

        function checkFaint() {
            const curP = party[activeIndex];

            if (enemy.hp <= 0 && !isBattleEnd) {
                isBattleEnd = true;
                playSE('win');
                addLog(`<span class='log-win'>戦闘勝利！ 野生の ${enemy.name} を倒した！</span>`);
                endBattle();
                return;
            }

            if (curP.hp <= 0) {
                addLog(`<span class='log-lose'>${curP.name} は倒れた！</span>`);
                const alive = party.filter(m => m.hp > 0);
                if (alive.length > 0) {
                    openSwitchMenu();
                } else {
                    isBattleEnd = true;
                    playSE('lose');
                    addLog(`<span class='log-lose'>全滅した... 敗北です。</span>`);
                    endBattle();
                }
            }
        }

        function openSwitchMenu() {
            els.commandArea.style.display = 'none';
            els.switchButtons.innerHTML = '';

            party.forEach((m, idx) => {
                if (m.hp > 0 && idx !== activeIndex) {
                    const btn = document.createElement('button');
                    btn.className = 'command-btn btn-switch';
                    btn.innerText = `${m.name} (HP:${m.hp}/${m.maxHp})`;
                    btn.onclick = () => switchMonster(idx);
                    els.switchButtons.appendChild(btn);
                }
            });

            els.switchArea.style.display = 'block';
        }

        function switchMonster(idx) {
            playSE('click');
            activeIndex = idx;
            lastPlayerCmd = null;
            els.switchArea.style.display = 'none';
            els.commandArea.style.display = 'block';
            addLog(`<span class='log-player'>ゆけっ！ ${party[activeIndex].name}！</span>`);
            updateUi();
        }

        function endBattle() {
            els.commandArea.style.display = 'none';
            els.magicSelectArea.style.display = 'none';
            els.switchArea.style.display = 'none';
            els.resetBtn.style.display = 'block';
        }

        // 初期化＆ループ開始
        renderCharacterGrid();
        requestAnimationFrame(mainLoop);
    </script>
